CREATE OR REPLACE FUNCTION clinic_finance.stock_consume_base(p_location uuid, p_item uuid, p_quantity numeric, p_kind text, p_note text DEFAULT ''::text, p_invoice text DEFAULT ''::text, p_payment text DEFAULT ''::text, p_lot uuid DEFAULT NULL::uuid)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare i public.inventory_items%rowtype; a record; v_id uuid;
begin
 if not public.inventory_location_can(p_location,'receive') and not public.inventory_location_can(p_location,'manage') then raise exception 'غير مسموح بالصرف من هذا المخزن'; end if;
 if p_kind not in ('issue','sale','waste','expired') then raise exception 'نوع حركة غير صحيح'; end if;
 if p_kind in ('waste','expired') and (not public.inventory_location_can(p_location,'manage') or length(trim(coalesce(p_note,'')))<2) then raise exception 'الهالك يحتاج مسؤولاً وسبباً'; end if;

  insert into public.inventory_movements(item_id,location_id,lot_id,movement_type,quantity_delta,unit_cost,unit_sale_price,margin_percent,profit_amount,patient_invoice_ref,payment_method,note,created_by)
   values(p_item,p_location,a.lot_id,p_kind,-a.qty,a.unit_cost,case when p_kind='sale' then i.current_sale_price else 0 end,i.markup_percent,
   case when p_kind='sale' then round((i.current_sale_price-a.unit_cost)*a.qty,2) when p_kind in ('waste','expired') then -round(a.unit_cost*a.qty,2) else 0 end,
   coalesce(p_invoice,''),coalesce(p_payment,''),coalesce(p_note,''),public.my_admin_id()) returning id into v_id;

######## clinic_finance.supplier_receive_base(p_id uuid, p_lines jsonb, p_note text)
CREATE OR REPLACE FUNCTION clinic_finance.supplier_receive_base(p_id uuid, p_lines jsonb, p_note text)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare req public.inventory_requests%rowtype; r jsonb; ln public.inventory_request_lines%rowtype; i public.inventory_items%rowtype; q numeric; cost numeric; price numeric; lid uuid;
begin
 if not public.inventory_location_can('00000000-0000-4000-8000-000000000001','manage') then raise exception 'التوريد لمسؤول المخزن الرئيسي'; end if;
 select * into req from public.inventory_requests where id=p_id for update;
 if not found or req.request_type<>'purchase' or req.status not in ('approved','partial') then raise exception 'طلب التوريد يحتاج موافقة'; end if;
 if jsonb_array_length(coalesce(p_lines,'[]'))=0 then raise exception 'أدخل الكميات'; end if;
 for r in select * from jsonb_array_elements(p_lines) order by value->>'item_id' loop
  select * into ln from public.inventory_request_lines where request_id=p_id and item_id=(r->>'item_id')::uuid for update;
  if not found then raise exception 'صنف خارج الطلب'; end if;
  q:=(r->>'quantity')::numeric;
  if q is null or q<=0 or ln.received_quantity+q>ln.quantity then raise exception 'الكمية تتجاوز المتبقي'; end if;
  select * into i from public.inventory_items where id=ln.item_id for update;
  cost:=coalesce((r->>'package_cost')::numeric,ln.package_cost)/i.units_per_pack;
  if cost is null or cost<0 then raise exception 'راجع تكلفة العبوة'; end if;
  if i.track_lots and (nullif(r->>'expiry_date','') is null or nullif(r->>'lot_number','') is null) then raise exception 'رقم التشغيلة والصلاحية مطلوبان'; end if;
  lid:=null;
  if i.track_lots or nullif(r->>'lot_number','') is not null or nullif(r->>'expiry_date','') is not null or nullif(r->>'production_date','') is not null then
   insert into public.inventory_lots(item_id,lot_number,expiry_date,received_quantity,unit_cost,supplier_name,created_by,production_date,shelf_life_months,lot_code_missing)
    values(i.id,coalesce(r->>'lot_number',''),nullif(r->>'expiry_date','')::date,q,cost,req.supplier_name,public.my_admin_id(),nullif(r->>'production_date','')::date,nullif(r->>'shelf_life_months','')::integer,coalesce((r->>'lot_code_missing')::boolean,false)) returning id into lid;
  end if;
  price:=round(greatest(cost,coalesce(clinic_finance.inventory_highest_purchase(i.id),0))*(1+i.markup_percent/100),2);
  if price<>i.current_sale_price then insert into public.inventory_price_history(item_id,previous_price,new_price,previous_markup,new_markup,changed_by) values(i.id,i.current_sale_price,price,i.markup_percent,i.markup_percent,public.my_admin_id()); end if;
  update public.inventory_items set current_sale_price=price,updated_at=now() where id=i.id;
  insert into public.inventory_movements(item_id,lot_id,movement_type,quantity_delta,unit_cost,unit_sale_price,margin_percent,reference_type,reference_id,note,created_by)
   values(i.id,lid,'receipt',q,cost,price,i.markup_percent,'purchase',p_id,coalesce(p_note,''),public.my_admin_id());
  update public.inventory_request_lines set received_quantity=received_quantity+q where id=ln.id;
 end loop;
 update public.inventory_requests set status=case when exists(select 1 from public.inventory_request_lines where request_id=p_id and received_quantity<quantity) then 'partial' else 'completed' end,completed_at=now() where id=p_id;
end $function$



######## clinic_reception.consume(p_workspace uuid, p_item uuid, p_location uuid, p_quantity numeric, p_kind text, p_source uuid)
CREATE OR REPLACE FUNCTION clinic_reception.consume(p_workspace uuid, p_item uuid, p_location uuid, p_quantity numeric, p_kind text, p_source uuid)
 RETURNS numeric
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'pg_catalog', 'public'
AS $function$
declare l record;remaining numeric:=p_quantity;q numeric;total numeric:=0;v uuid;price numeric;dims jsonb;
begin
 perform clinic_reception.number(p_quantity::text,3);
 if not exists(select 1 from clinic_reception.items where id=p_item and workspace_id=p_workspace) or not exists(select 1 from clinic_reception.locations where id=p_location and workspace_id=p_workspace) then raise exception 'الصنف أو المخزن خارج المساحة';end if;
 select sale_price into price from clinic_reception.items where id=p_item;
 for l in select lot.id,lot.unit_cost,sum(m.quantity) quantity from clinic_reception.lots lot join clinic_reception.movements m on m.lot_id=lot.id where lot.workspace_id=p_workspace and lot.item_id=p_item and m.location_id=p_location and (p_kind='waste' or lot.expiry_date is null or lot.expiry_date>=clinic_reception.today()) group by lot.id having sum(m.quantity)>0 order by lot.expiry_date nulls last,lot.received_at,lot.id loop
  q:=least(remaining,l.quantity);insert into clinic_reception.movements(workspace_id,item_id,location_id,lot_id,kind,quantity,unit_cost,sale_price,source_id,actor_id) values(p_workspace,p_item,p_location,l.id,p_kind,-q,l.unit_cost,case when p_kind='sale' then price else 0 end,p_source,public.my_admin_id()) returning id into v;
  dims:=jsonb_build_object('order_id',p_source,'item_id',p_item,'location_id',p_location);
  if p_kind in ('included','sale','waste') then perform clinic_reception.post(p_workspace,'stock_cost',v,'تكلفة صرف مستلزمات',case p_kind when 'included' then '5123' when 'sale' then '5121' else '5217' end,'1131',round(q*l.unit_cost,2),dims);end if;
  total:=total+round(q*l.unit_cost,2);remaining:=remaining-q;exit when remaining=0;
 end loop;
 if remaining>0 then raise exception 'المخزون المتاح غير كاف أو منتهي الصلاحية';end if;return total;
end $function$

 expected=clinic_finance.shift_cash(f.id),
 difference=f.counted-clinic_finance.shift_cash(f.id)
 from clinic_finance.shifts f where mapped.workspace_id=p_workspace and mapped.finance_shift_id=f.id;
 insert into clinic_reception.shifts(id,workspace_id,register_id,cashier_id,status,opened_at,closed_at,counted,finance_shift_id)
 select f.id,p_workspace,r.id,f.cashier_id,f.status,f.opened_at,f.closed_at,f.counted,f.id from clinic_finance.shifts f join clinic_reception.registers r on r.finance_register_id=f.register_id and r.workspace_id=p_workspace
 where not exists(select 1 from clinic_reception.shifts mapped where mapped.workspace_id=p_workspace and mapped.finance_shift_id=f.id)
 on conflict(id) do nothing;
end $function$



######## clinic_reception.sync_stock(p_workspace uuid)
CREATE OR REPLACE FUNCTION clinic_reception.sync_stock(p_workspace uuid)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'pg_catalog', 'public'
AS $function$
declare x record;l uuid;current_qty numeric;highest numeric;
begin
 if not exists(select 1 from clinic_reception.workspaces where id=p_workspace and mode='live') then return;end if;
 -- Match by immutable IDs only. Never infer a mapping from a name or phone.
 perform i.id from public.inventory_items i join clinic_reception.items rx on rx.inventory_item_id=i.id where rx.workspace_id=p_workspace order by i.id for update of i;
 for x in select rx.id item_id,loc.id location_id,l.id inventory_lot_id,l.lot_number,l.expiry_date,l.unit_cost,l.received_at,coalesce(sum(m.quantity_delta),0) quantity from clinic_reception.items rx join public.inventory_lots l on l.item_id=rx.inventory_item_id cross join clinic_reception.locations loc left join public.inventory_movements m on m.lot_id=l.id and m.location_id=loc.inventory_location_id where rx.workspace_id=p_workspace and loc.workspace_id=p_workspace and loc.inventory_location_id is not null group by rx.id,loc.id,l.id loop
  select id into l from clinic_reception.lots where workspace_id=p_workspace and inventory_lot_id=x.inventory_lot_id;
  if l is null then insert into clinic_reception.lots(id,workspace_id,item_id,number,expiry_date,unit_cost,inventory_lot_id,received_at) values(x.inventory_lot_id,p_workspace,x.item_id,x.lot_number,x.expiry_date,x.unit_cost,x.inventory_lot_id,x.received_at) returning id into l;end if;
  select coalesce(sum(quantity),0) into current_qty from clinic_reception.movements where workspace_id=p_workspace and lot_id=l and location_id=x.location_id;
  if current_qty<>x.quantity then insert into clinic_reception.movements(workspace_id,item_id,location_id,lot_id,kind,quantity,unit_cost,source_id,actor_id) values(p_workspace,x.item_id,x.location_id,l,'count',x.quantity-current_qty,x.unit_cost,null,public.my_admin_id());end if;
 end loop;
 -- Highest purchase cost is a pricing reference, not a retrospective cost revaluation.
 update clinic_reception.items rx set highest_cost=greatest(rx.highest_cost,coalesce((select max(unit_cost) from public.inventory_lots where item_id=rx.inventory_item_id),0)),sale_price=round(greatest(rx.highest_cost,coalesce((select max(unit_cost) from public.inventory_lots where item_id=rx.inventory_item_id),0))*(1+rx.markup/100),2) where rx.workspace_id=p_workspace and rx.inventory_item_id is not null;
end $function$

######## public.inventory_create_transfer(p_source uuid, p_destination uuid, p_receiver uuid, p_lines jsonb, p_note text)

######## public.inventory_dispense_patient(p_payload jsonb, p_token uuid)
CREATE OR REPLACE FUNCTION public.inventory_dispense_patient(p_payload jsonb, p_token uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'pg_catalog', 'public'
AS $function$
declare w uuid;me uuid;prev clinic_reception.commands%rowtype;vis clinic_reception.visits%rowtype;
 patient clinic_reception.patients%rowtype;inv clinic_reception.invoices%rowtype;it public.inventory_items%rowtype;

 if p_token is null or jsonb_typeof(p_payload) is distinct from 'object' or octet_length(p_payload::text)>6000 then raise exception 'طلب غير صالح';end if;
 w:=nullif(p_payload->>'workspace_id','')::uuid;

 perform clinic_finance.lock_day();

 if not found then raise exception 'اختر زيارة مفتوحة للمريض؛ الزيارة المغلقة لا تستقبل صرفًا جديدًا';end if;

 if not found then raise exception 'الزيارة لا تخص ملف المريض المختار';end if;

######## public.inventory_location_can(p_location uuid, p_action text)
CREATE OR REPLACE FUNCTION public.inventory_location_can(p_location uuid, p_action text)
 RETURNS boolean
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
 select exists(select 1 from public.admins a where a.id=public.my_admin_id() and a.active)
 and exists(select 1 from public.inventory_locations l where l.id=p_location and l.active)
 and case when p_action='storekeeper' then exists(select 1 from public.inventory_location_staff where location_id=p_location and admin_id=public.my_admin_id() and responsibility='storekeeper')
 when public.inventory_is_manager() then p_action in ('view','manage','request','receive','count','manager')
 else exists(select 1 from public.inventory_location_staff s where s.location_id=p_location and s.admin_id=public.my_admin_id() and
  case p_action when 'view' then true when 'manage' then s.responsibility in ('storekeeper','manager')
   when 'request' then s.responsibility='senior_nursing' when 'receive' then s.responsibility in ('receiver','senior_nursing','storekeeper')
   when 'count' then s.responsibility in ('counter','senior_nursing','storekeeper','manager')
   when 'manager' then s.responsibility='manager' else false end) end;
$function$

######## public.inventory_receive_transfer(p_id uuid, p_lines jsonb, p_note text)
CREATE OR REPLACE FUNCTION public.inventory_receive_transfer(p_id uuid, p_lines jsonb, p_note text DEFAULT ''::text)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare t public.inventory_transfers%rowtype; r record; q numeric; mismatch boolean:=false;
begin
 select * into t from public.inventory_transfers where id=p_id for update;
 if not found or t.status<>'in_transit' then raise exception 'التحويل غير متاح للاستلام'; end if;
 if t.receiver_id<>public.my_admin_id() or not public.inventory_location_can(t.destination_id,'receive') then raise exception 'تأكيد الاستلام للمستلم المحدد'; end if;
 if jsonb_typeof(p_lines)<>'array' or jsonb_array_length(p_lines)<>(select count(*) from public.inventory_transfer_allocations a join public.inventory_transfer_lines l on l.id=a.line_id where l.transfer_id=p_id) or (select count(distinct x->>'allocation_id') from jsonb_array_elements(p_lines) x)<>jsonb_array_length(p_lines) then raise exception 'أدخل كل كميات التشغيلات المستلمة دون تكرار'; end if;
 for r in select a.*,l.item_id from public.inventory_transfer_allocations a join public.inventory_transfer_lines l on l.id=a.line_id where l.transfer_id=p_id order by l.item_id,a.id loop
  select (x->>'quantity')::numeric into q from jsonb_array_elements(p_lines) x where (x->>'allocation_id')::uuid=r.id;
  if q is null or q<0 or q>r.quantity then raise exception 'كمية استلام غير صحيحة'; end if;
  perform 1 from public.inventory_items where id=r.item_id for update;
  update public.inventory_transfer_allocations set received_quantity=q where id=r.id;
  if q>0 then insert into public.inventory_movements(item_id,location_id,lot_id,movement_type,quantity_delta,unit_cost,reference_type,reference_id,created_by,note)
   values(r.item_id,t.destination_id,r.lot_id,'transfer_in',q,r.unit_cost,'transfer',p_id,public.my_admin_id(),coalesce(p_note,'')); end if;
  mismatch:=mismatch or q<>r.quantity;
 end loop;
 if mismatch and length(trim(coalesce(p_note,'')))<2 then raise exception 'اكتب ملاحظة لفروق الاستلام'; end if;
 update public.inventory_transfers set status=case when mismatch then 'discrepancy' else 'received' end,received_by=public.my_admin_id(),received_at=now() where id=p_id;
 insert into public.inventory_audit_events(entity_id,kind,actor_id,detail) values(p_id,'transfer_received',public.my_admin_id(),coalesce(p_note,''));
end $function$
